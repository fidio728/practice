. program define post_twfe_coef
  1.     args posth comparison term coefname n firms treated control
  2.     tempname bb vv
  3.     scalar `bb' = .
  4.     scalar `vv' = .
  5.     capture scalar `bb' = _b[`coefname']
  6.     capture scalar `vv' = _se[`coefname']
  7.     scalar pp = 2 * ttail(e(df_r), abs(`bb' / `vv'))
  8.     post `posth' ("`comparison'") ("TWFE") ("`term'") (`bb') (`vv') (pp) (`n') (`firms') (`treated') (`control')
  9. end

. program define post_bjs_tau
  1.     args posth comparison term n firms treated control
  2.     tempname bb vv
  3.     scalar `bb' = .
  4.     scalar `vv' = .
  5.     capture scalar `bb' = _b[tau]
  6.     if _rc {

 10.     if _rc {
