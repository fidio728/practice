\begin{frame}[t]{Essay 1: crawler access and disclosure-window liquidity}
\eyebrow{Current paper --- short update}

\begin{columns}[T,onlytextwidth]
  \begin{column}{0.53\textwidth}
    \smallhead{Question and design}
    {\bodytext
    \begin{itemize}\setlength\itemsep{0.15cm}
      \item Do firm-initiated crawler restrictions change how markets process mandatory filings?
      \item Treatment: first restriction of any of 31 search/discovery crawlers in archived \texttt{robots.txt} files.
      \item Staggered design, 2021--2025: 27,327 10-K/10-Q events; 1,908 firms.
      \item Outcome: filing-window log Amihud minus its local benchmark.
    \end{itemize}}
  \end{column}
  \begin{column}{0.43\textwidth}
    \smallhead{Current result}
    \metric{+0.0134}{Abnormal log Amihud (BJS), $p=0.010$}
    \vspace{0.18cm}
    {\smalltext
    \begin{itemize}\setlength\itemsep{0.11cm}
      \item Benchmark liquidity improves faster; the filing window does not keep pace.
      \item Quoted spreads: null; dollar volume: up.
      \item Return variance: up; post-filing drift: null.
      \item \textbf{Reading:} price impact/depth, not trading withdrawal.
    \end{itemize}}
  \end{column}
\end{columns}
\end{frame}

\begin{frame}[t]{Essay 1: Choosing to Become Harder to Find}
\eyebrow{Measurement + market microstructure}

\begin{columns}[T,onlytextwidth]
  \begin{column}{0.48\textwidth}
    \smallhead{Research question}
    {\bodytext Do firms restrict crawler access to their websites, and does this access choice change trading around mandatory disclosures?}\par
    \vspace{0.22cm}
    \smallhead{Core evidence}
    {\bodytext
    \begin{itemize}\setlength\itemsep{0.12cm}
      \item New monthly panel of crawler-access policies and staggered switches.
      \item Filing-window price impact relative to local market conditions.
    \end{itemize}}
  \end{column}
  \begin{column}{0.48\textwidth}
    \smallhead{Contribution}
    {\bodytext
    \begin{itemize}\setlength\itemsep{0.12cm}
      \item The supply of discoverability is a firm choice, not a fixed feature of public disclosure.
      \item Information can remain legally public while becoming economically harder to find.
      \item The access choice has a measurable market-microstructure consequence.
    \end{itemize}}
    \vspace{0.14cm}
    \colorbox{panel}{\parbox{0.93\linewidth}{\smalltext\textbf{Boundary:} this is not an AI paper, and it does not equate illiquidity with informational inefficiency.}}
  \end{column}
\end{columns}

\end{frame}

\begin{frame}[t]{Essay 2: External monitoring}
\eyebrow{Turning Off the Machines --- access and the erosion of monitoring}

\begin{columns}[T,onlytextwidth]
  \begin{column}{0.39\textwidth}
    \smallhead{Core question}
    {\bodytext Can a firm weaken outside monitoring by withdrawing machine access to information that remains publicly available?}\par
    \vspace{0.20cm}
    \smallhead{Mechanism}
    {\bodytext Machine discovery is an input into scalable monitoring. Blocking raises detection costs and lowers the probability that irregular behavior is noticed.}\par
    \vspace{0.10cm}
    \colorbox{accentlight}{\parbox{0.92\linewidth}{\smalltext\bfseries New object: monitoring technology, not trading liquidity.}}
  \end{column}
  \begin{column}{0.57\textwidth}
    \smallhead{Empirical design}
    {\smalltext
    \begin{itemize}\setlength\itemsep{0.11cm}
      \item Access restrictions $\rightarrow$ managerial outcomes; use passive/vendor-driven adoption where possible.
      \item Heterogeneity by machine-information reliance and conventional monitoring.
      \item Placebos: routine insider trades; firms with strong institutional monitoring.
    \end{itemize}}
    \vspace{0.12cm}
    \smallhead{Primary outcome families}
    {\smalltext
    \begin{itemize}\setlength\itemsep{0.10cm}
      \item Opportunistic Form 4 trading and trading profitability.
      \item Discretionary accruals and real earnings management.
      \item Late filings, amendments, restatements, and reporting discipline.
    \end{itemize}}
  \end{column}
\end{columns}
\end{frame}

\vspace{0.27cm}
\begin{columns}[T,onlytextwidth]
  \begin{column}{0.48\textwidth}
    \smallhead{Temporal reallocation}
    {\bodytext Pre-filing response falls, filing response rises, and the full-cycle response is unchanged. Information is delayed, not destroyed.}
  \end{column}
  \begin{column}{0.48\textwidth}
    \smallhead{Information destruction}
    {\bodytext The filing response does not fully compensate; total price--fundamental covariance falls or post-filing learning persists.}
  \end{column}
\end{columns}

\vspace{0.08cm}
{\smalltext\bfseries\color{accent}Main requirement: link returns to future fundamentals. Variance alone shows when prices move, not when they learn.}

\end{frame}

\begin{frame}[t]{When is Essay 3 a separate paper?}
\eyebrow{Overlap audit against Essay 1}

\begin{columns}[T,onlytextwidth]
  \begin{column}{0.47\textwidth}
    {\color{red}\fontsize{12}{14}\selectfont\bfseries MERGE INTO ESSAY 1 IF}\par\vspace{0.15cm}
    {\bodytext
    \begin{itemize}\setlength\itemsep{0.15cm}
      \item It uses the same robots event, sample, filing window, and staggered event study.
      \item Price delay, variance ratios, synchronicity, or announcement volatility are simply added as outcomes.
      \item The conclusion is only that restrictions worsen market quality.
    \end{itemize}}
    \vspace{0.22cm}
    \metric{70--80\%}{Estimated design overlap in this weak version}
  \end{column}
  \begin{column}{0.04\textwidth}\centering
    {\color{rule}\rule{0.6pt}{4.45cm}}
  \end{column}
  \begin{column}{0.47\textwidth}
    {\color{green}\fontsize{12}{14}\selectfont\bfseries STANDALONE IF}\par\vspace{0.15cm}
    {\bodytext
    \begin{itemize}\setlength\itemsep{0.15cm}
      \item The estimand is the \textbf{timing versus total amount} of fundamental learning.
      \item The unit is the inter-filing learning cycle, not filing-window illiquidity.
      \item Returns are anchored to future earnings or another external fundamental realization.
      \item Preferably, identification comes from a separate external shock to search discovery.
    \end{itemize}}
  \end{column}
\end{columns}

\vspace{0.10cm}
{\smalltext\bfseries\color{accent}Strongest version: ask when prices learn, not whether trading becomes less liquid.}
\end{frame}
